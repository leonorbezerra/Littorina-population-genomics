data_pi <- inner_join(x_pi, y_pi, by = c("CHROM", "BIN_START")) %>%
  inner_join(t_pi, by = c("CHROM", "BIN_START")) %>%
  mutate(
    # Aplicação exata da fórmula matemática isolada para dxy
    dxy = (NT*(NT-1)*pi_t - NX*(NX-1)*pi_x - NY*(NY-1)*pi_y) / (2 * NX * NY),
        
    # Conversão para Mb para os eixos do gráfico
    pos = BIN_START / 1e6,
    region = if_else(pos >= 37 & pos <= 52, "Inv", "Col")
  )

# Cria a tabela longa para o GRÁFICO 1
pi <- data_pi %>%
  select(CHROM, BIN_START, region, !!GROUP_AA := pi_x, !!GROUP_BB := pi_y) %>%
  pivot_longer(cols = c(!!sym(GROUP_AA), !!sym(GROUP_BB)), names_to = "group", values_to = "pi")



# ---- WILCOXON TESTS +  BONFERRONI (HOLM) ----

wilcox_inv <- data_pi %>% filter(region == "Inv")
wilcox_col   <- data_pi %>% filter(region == "Col")

p_values <- c(
  t1 = wilcox.test(pi_x ~ region, data = data_pi)$p.value,
  t2 = wilcox.test(pi_y ~ region, data = data_pi)$p.value,
  t3 = wilcox.test(dxy ~ region, data = data_pi)$p.value,
  t4 = wilcox.test(wilcox_inv$pi_x, wilcox_inv$pi_y, paired = TRUE)$p.value,
  t5 = wilcox.test(wilcox_col$pi_x, wilcox_col$pi_y, paired = TRUE)$p.value
)
p_corrected <- p.adjust(p_values, method = "holm")

tests <- c(
  "1. Pi Group AA — Inv) vs Col",
  "2. Pi Group BB — Inv) vs Col",
  "3. dXY — Inv vs Col",
  "4. Inv — Group AA vs Group BB",
  "5. Col — Group AA vs Group BB"
)

sink("wilcoxon_tests_pi.txt")

for(i in 1:length(p_values)) {
  cat(tests[i], "\n")
  cat("  -> P-value:     ", format.pval(p_values[i], digits = 5), "\n")
  cat("  -> P-value Corrected: ", format.pval(p_corrected[i], digits = 5), "\n")
  if(p_corrected[i] < 0.05) {
    cat("  -> Result:          Significative (p < 0.05)\n")
  } else {
    cat("  -> Result:          Not Significative (p >= 0.05)\n")
  }
  cat("\n-------------------------------\n\n")
}

chi2 <- qchisq(1 - p_values, df = 1)
lambda <- median(chi2, na.rm = TRUE) / qchisq(0.5, df = 1)
cat("Genomic inflation lambda:", lambda, "\n")
observed_log <- -log10(sort(p_values))
expected_log <- -log10(ppoints(length(p_values)))
plot(expected_log, observed_log,
     main = paste("Gemma QQ-Plot, lambda=", round(lambda, 2), ")"),
     xlab = "Waited (-log10 p)",
     ylab = "Observed (-log10 p)",
     pch = 20, col = "darkblue")
abline(0, 1, col = "red", lwd = 2)
